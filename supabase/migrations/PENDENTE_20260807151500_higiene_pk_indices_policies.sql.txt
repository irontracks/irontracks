-- ── 1. device_push_tokens sem PRIMARY KEY ────────────────────────────────
-- Já existe UNIQUE (token), sem nulo nem duplicata (37/37 conferido). A
-- UNIQUE vira PK: mesma coluna, mesma garantia, e a tabela deixa de ser
-- "sem identidade" para replicação e ferramentas.
alter table public.device_push_tokens
  drop constraint if exists device_push_tokens_token_unique;
alter table public.device_push_tokens
  add constraint device_push_tokens_pkey primary key (token);

-- ── 2. student_diet_plans: policies permissivas duplicadas ───────────────
-- A policy ALL (`own_write`) se sobrepunha às quatro de admin, então CADA
-- query avaliava duas policies. Consolidado em uma por ação, com o mesmo OR
-- que o Postgres já fazia entre permissivas — a semântica NÃO muda:
--   admin → tudo;  dono → só as próprias linhas (user_id = created_by = uid).
-- O SELECT já estava consolidado (`student_diet_plans_select__merged`) e não
-- é tocado aqui.
drop policy if exists student_diet_plans_own_write    on public.student_diet_plans;
drop policy if exists student_diet_plans_insert_admin on public.student_diet_plans;
drop policy if exists student_diet_plans_update_admin on public.student_diet_plans;
drop policy if exists student_diet_plans_delete_admin on public.student_diet_plans;

create policy student_diet_plans_insert on public.student_diet_plans
  for insert to authenticated
  with check (
    (select is_admin())
    or (user_id = (select auth.uid()) and created_by = (select auth.uid()))
  );

create policy student_diet_plans_update on public.student_diet_plans
  for update to authenticated
  using (
    (select is_admin())
    or (user_id = (select auth.uid()) and created_by = (select auth.uid()))
  )
  with check (
    (select is_admin())
    or (user_id = (select auth.uid()) and created_by = (select auth.uid()))
  );

create policy student_diet_plans_delete on public.student_diet_plans
  for delete to authenticated
  using (
    (select is_admin())
    or (user_id = (select auth.uid()) and created_by = (select auth.uid()))
  );
